terraform {
  required_version = ">= 1.5.0"

  required_providers {
    aws = {
      source  = "hashicorp/aws"
      version = ">= 5.0"
    }
  }
}

############################################
# Providers
############################################

# Tokyo (ap-northeast-1) — all VPC/EC2/RDS/ALB stuff
provider "aws" {
  alias  = "tokyo"
  region = "ap-northeast-1"
}

# us-east-1 — REQUIRED for ACM cert used by CloudFront + CloudFront resources
provider "aws" {
  alias  = "us_east_1"
  region = "us-east-1"
}

############################################
# Tokyo Authority Module
############################################
module "tokyo_authority" {
  source = "./modules/tokyo_authority"

  providers = {
    aws          = aws.tokyo
    aws.us_east_1 = aws.us_east_1
  }

  # ---- required inputs (fill these based on your module variables) ----

  project_name = var.project_name

  # DNS / CloudFront
  domain_name  = var.domain_name          # e.g. "keyescloudsolutions.com"
  app_subdomain = var.app_subdomain       # e.g. "www" (optional if your module assumes www)

  # EC2
  tokyo_ami_id       = var.tokyo_ami_id   # or ami_id, depending on your module var names
  instance_type      = var.instance_type  # e.g. "t3.micro"

  # RDS (if your module wants these)
  db_name     = var.db_name
  db_username = var.db_username
  db_password = var.db_password

  # If your module already creates the VPC/subnets internally, you do NOT pass vpc_id/subnet_ids.
  # If your module expects existing networking, you WOULD pass them here.
  # vpc_id            = var.tokyo_vpc_id
  # public_subnet_ids  = var.tokyo_public_subnet_ids
  # private_subnet_ids = var.tokyo_private_subnet_ids
}

############################################
# Root Outputs (optional convenience)
############################################
output "tokyo_cloudfront_domain_name" {
  value = module.tokyo_authority.cloudfront_domain_name
}

output "tokyo_app_fqdn" {
  value = module.tokyo_authority.app_fqdn
}
